// ============================================================

// BMI270底层寄存器写

// ============================================================

  

bool bmiWriteRegister(

    uint8_t reg,

    uint8_t value

)

{

    BMIWire.beginTransmission(

        BMI270_ADDRESS

    );

  

    BMIWire.write(reg);

    BMIWire.write(value);

  

    uint8_t error =

        BMIWire.endTransmission();

  

    delayMicroseconds(500);

  

    return

        error == 0;

}

  
  

// ============================================================

// BMI270底层寄存器读

// ============================================================

  

bool bmiReadRegister(

    uint8_t reg,

    uint8_t &value

)

{

    BMIWire.beginTransmission(

        BMI270_ADDRESS

    );

  

    BMIWire.write(reg);

  

    if (

        BMIWire.endTransmission(false)

        != 0

    )

    {

        return false;

    }

  
  

    uint8_t count =

        BMIWire.requestFrom(

            BMI270_ADDRESS,

            (uint8_t)1

        );

  
  

    if (

        count != 1

        ||

        !BMIWire.available()

    )

    {

        return false;

    }

  
  

    value =

        BMIWire.read();

  
  

    return true;

}

  
  

// ============================================================

// 配置BMI270为400Hz

// ============================================================

  

bool configureBMI270_400Hz()

{

    if (

        !bmiWriteRegister(

            BMI270_REG_ACC_CONF,

            BMI270_ACC_CONF_400HZ

        )

    )

    {

        return false;

    }

  
  

    delay(5);

  
  

    if (

        !bmiWriteRegister(

            BMI270_REG_GYR_CONF,

            BMI270_GYR_CONF_400HZ

        )

    )

    {

        return false;

    }

  
  

    delay(10);

  
  

    // 回读确认

    uint8_t accConf = 0;

    uint8_t gyrConf = 0;

  
  

    if (

        !bmiReadRegister(

            BMI270_REG_ACC_CONF,

            accConf

        )

    )

    {

        return false;

    }

  
  

    if (

        !bmiReadRegister(

            BMI270_REG_GYR_CONF,

            gyrConf

        )

    )

    {

        return false;

    }

  
  

    return

        accConf == BMI270_ACC_CONF_400HZ

        &&

        gyrConf == BMI270_GYR_CONF_400HZ;

}

  
  

// ============================================================

// BMI270读取

// ============================================================

  

bool readBMI270()

{

    if (

        imu.getSensorData()

        != BMI2_OK

    )

    {

        return false;

    }

  
  

    ax = imu.data.accelX;

    ay = imu.data.accelY;

    az = imu.data.accelZ;

  

    gx = imu.data.gyroX;

    gy = imu.data.gyroY;

    gz = imu.data.gyroZ;

  
  

    return true;

}

        )

        {

            delay(200);

                imu.getSensorData()

                ==

                BMI2_OK

            )

            {

                return true;

            }

        }

  
  

        delay(300);

    }

  
  

    return false;

}

        PI;

// ============================================================

// IMU校准

// 开机保持车体静止

// ============================================================

  

bool calibrateIMU()

{

    const int SAMPLE_COUNT = 500;

  

    float pitchSum = 0.0f;

    float gyroYSum = 0.0f;

    float gyroXSum = 0.0f;

  

    int validCount = 0;

  
  

    delay(500);

  
  

    for (

        int i = 0;

        i < SAMPLE_COUNT;

        i++

    )

    {

        if (

            readBMI270()

        )

        {

            pitchSum +=

                calculateAccelPitch();

  

            gyroYSum +=

                gy;

  

            gyroXSum +=

                gx;

  

            validCount++;

        }

  
  

        delay(3);

    }

  
  

    if (

        validCount

        <

        SAMPLE_COUNT / 2

    )

    {

        return false;

    }

  
  

    pitchZero =

        pitchSum

        /

        validCount;

  
  

    gyroYZero =

        gyroYSum

        /

        validCount;

  
  

    gyroXZero =

        gyroXSum

        /

        validCount;

  
  

    accPitch = 0.0f;

    pitch = 0.0f;

  

    kalmanAngle = 0.0f;

    kalmanBias = 0.0f;

  

    P00 = 1.0f;

    P01 = 0.0f;

    P10 = 0.0f;

    P11 = 1.0f;

  
  

    return true;

}

  
  

// ============================================================

// Kalman

// ============================================================

  

float updateKalman(

    float measuredAngle,

    float measuredRate,

    float dt

)

{

    // --------------------------------------------------------

    // 1. 状态预测

    // --------------------------------------------------------

  

    float rate =

        measuredRate

        -

        kalmanBias;

  
  

    kalmanAngle +=

        dt

        *

        rate;

  
  

    // --------------------------------------------------------

    // 2. 协方差预测

    // --------------------------------------------------------

  

    P00 +=

        dt

        *

        (

            dt * P11

            -

            P01

            -

            P10

            +

            Q_angle

        );

  
  

    P01 -=

        dt

        *

        P11;

  
  

    P10 -=

        dt

        *

        P11;

  
  

    P11 +=

        Q_bias

        *

        dt;

  
  

    // --------------------------------------------------------

    // 3. 测量创新

    // --------------------------------------------------------

  

    float innovation =

        measuredAngle

        -

        kalmanAngle;

  
  

    // --------------------------------------------------------

    // 动态测量噪声

    //

    // 车辆静止/低动态时 dynamicR ≈ 0.03

    // 车辆加速或振动明显时 dynamicR 增大，

    // 从而暂时减少加速度角对Kalman的修正。

    // --------------------------------------------------------

  

    float S =

        P00

        +

        dynamicR;

  
  

    // --------------------------------------------------------

    // 4. Kalman增益

    // --------------------------------------------------------

  

    float K0 =

        P00

        /

        S;

  
  

    float K1 =

        P10

        /

        S;

  
  

    // --------------------------------------------------------

    // 5. 状态修正

    // --------------------------------------------------------

  

    kalmanAngle +=

        K0

        *

        innovation;

  
  

    kalmanBias +=

        K1

        *

        innovation;

  
  

    // --------------------------------------------------------

    // 6. 协方差修正

    // --------------------------------------------------------

  

    float oldP00 = P00;

    float oldP01 = P01;

  
  

    P00 -=

        K0

        *

        oldP00;

  
  

    P01 -=

        K0

        *

        oldP01;

  
  

    P10 -=

        K1

        *

        oldP00;

  
  

    P11 -=

        K1

        *

        oldP01;

  
  

    return

        kalmanAngle;

}

  
  

// ============================================================

// 姿态更新

// ============================================================

  

void updateAttitude()

{

    // --------------------------------------------------------

    // 1. 原始加速度角

    // --------------------------------------------------------

  

    accPitch =

        calculateAccelPitch()

        -

        pitchZero;

  
  

    // --------------------------------------------------------

    // 2. Pitch轴陀螺仪角速度

    // --------------------------------------------------------

  

    gyroPitchRate =

        GY_DIRECTION

        *

        (

            gy

            -

            gyroYZero

        );

  
  

    // --------------------------------------------------------

    // 3. 计算三轴加速度模长

    //

    // BMI270库这里的加速度量按g量级使用。

    // 静止时理论上接近1g。

    // --------------------------------------------------------

  

    accelMagnitude =

        sqrtf(

            ax * ax

            +

            ay * ay

            +

            az * az

        );

  
  

    accelMagnitudeError =

        fabsf(

            accelMagnitude

            -

            1.0f

        );

  
  

    // --------------------------------------------------------

    // 4. 根据 |a|-1g 动态调整R

    //

    // 越偏离1g，说明线性加速度/振动越明显，

    // 这时accPitch可信度降低。

    // --------------------------------------------------------

  

    if (

        accelMagnitudeError

        <

        ACC_ERR_LOW

    )

    {

        dynamicR =

            R_LOW_DYNAMIC;

    }

    else if (

        accelMagnitudeError

        <

        ACC_ERR_HIGH

    )

    {

        dynamicR =

            R_MID_DYNAMIC;

    }

    else

    {

        dynamicR =

            R_HIGH_DYNAMIC;

    }

  
  

    // --------------------------------------------------------

    // 5. Kalman融合

    // --------------------------------------------------------

  

    pitch =

        updateKalman(

            accPitch,

            gyroPitchRate,

            CONTROL_DT

        );

  
  

    // --------------------------------------------------------

    // 6. 根据Pitch去除Z轴重力分量，得到前向线加速度趋势

    //

    // 静止时：

    // az ≈ -sin(pitch)

    //

    // 因此：

    // az + sin(pitch) ≈ 0

    //

    // 再乘方向参数统一为“向前加速为正”。

    // --------------------------------------------------------

  

    float pitchRad =

        pitch

        *

        DEG_TO_RAD;

  
  

    imuForwardAccelG =

        IMU_FORWARD_ACCEL_DIRECTION

        *

        (

            az

            +

            sinf(pitchRad)

        );

  
  

    // 尖峰限幅

    if (

        imuForwardAccelG

        >

        IMU_ACCEL_CLAMP_G

    )

    {

        imuForwardAccelG =

            IMU_ACCEL_CLAMP_G;

    }

  
  

    if (

        imuForwardAccelG

        <

        -IMU_ACCEL_CLAMP_G

    )

    {

        imuForwardAccelG =

            -IMU_ACCEL_CLAMP_G;

    }

  
  

    // 轻度低通

    imuForwardAccelFilteredG =

        (1.0f - IMU_ACCEL_LPF_ALPHA)

        *

        imuForwardAccelFilteredG

        +

        IMU_ACCEL_LPF_ALPHA

        *

        imuForwardAccelG;

  
  

    // 根据加速度模长偏离1g的程度降低IMU预测可信度

    if (

        accelMagnitudeError

        <

        ACC_ERR_LOW

    )

    {

        imuSpeedPredictionTrust =

            1.0f;

    }

    else if (

        accelMagnitudeError

        <

        ACC_ERR_HIGH

    )

    {

        imuSpeedPredictionTrust =

            0.45f;

    }

    else

    {

        imuSpeedPredictionTrust =

            0.12f;

    }

  
  

    // --------------------------------------------------------

    // 7. 转向角速度

    // --------------------------------------------------------

  

    gyroTurnRate =

        TURN_GYRO_DIRECTION

        *

        (

            gx

            -

            gyroXZero

        );

}

void setMotor1(

    float commandHz

)

{

    if (

        !isfinite(commandHz)

    )

    {

        return;

    }

  

    if (

        fabsf(commandHz)

        <

        MOTOR_STOP_THRESHOLD_HZ

    )

    {

        if (

            HOLD_LAST_OUTPUT_ON_ZERO

            &&

            motor1ZeroHoldCount

            <

            ZERO_HOLD_MAX_CYCLES

        )

        {

            motor1ZeroHoldCount++;

            return;

        }

  

        motor1ZeroHoldCount =

            0;

  

        stopMotor1();

        return;

    }

  

    // 新有效输出到来，清零0输出保持计数

    motor1ZeroHoldCount =

        0;

  
  

    bool dir =

        commandHz

        >=

        0.0f;

  
  

    if (

        motor1Invert

    )

    {

        dir =

            !dir;

    }

  
  

    bool directionChanged =

        (

            motor1LastCommandHz

            *

            commandHz

        )

        <

        0.0f;

  
  

    if (

        directionChanged

    )

    {

        motor1Timer.pause();

    }

  
  

    digitalWrite(

        MOTOR1_DIR_PIN,

        dir ? HIGH : LOW

    );

  
  

    if (

        directionChanged

    )

    {

        delayMicroseconds(2);

    }

  
  

    float frequency =

        fabsf(

            commandHz

        );

  
  

    bool frequencyChanged =

        motor1LastCommandHz

        ==

        0.0f

        ||

        fabsf(

            fabsf(

                motor1LastCommandHz

            )

            -

            frequency

        )

        >=

        PWM_UPDATE_MIN_DELTA_HZ;

  
  

    if (

        directionChanged

        ||

        frequencyChanged

    )

    {

        motor1Timer.setPWM(

            1,

            MOTOR1_STEP_PIN,

            (uint32_t)frequency,

            50

        );

  
  

        motor1Timer.resume();

  
  

        motor1LastCommandHz =

            commandHz;

    }

}

// ============================================================

// 多源速度Kalman

//

// 400Hz：用IMU前向线加速度做预测

// 20Hz ：用编码器速度做测量修正

//

// 电机命令不直接当“真实速度”，只用来动态调整过程噪声Q：

// 控制命令大 -> 系统更可能快速改变速度 -> Kalman更允许状态变化。

// ============================================================

  

void predictVehicleSpeedKalman()

{

    // --------------------------------------------------------

    // 1. IMU只负责“短期趋势预测”

    //

    // 编码器负责长期速度绝对值；

    // IMU不允许长期主导vehicleSpeed。

    // --------------------------------------------------------

  

    float accelForSpeed =

        imuForwardAccelFilteredG;

  
  

    // 小加速度死区：

    // 防止0.01~0.02g级别的零偏不断积分成假速度。

    if (

        fabsf(accelForSpeed)

        <

        IMU_SPEED_ACCEL_DEADBAND_G

    )

    {

        accelForSpeed =

            0.0f;

    }

  
  

    float imuDeltaSpeed =

        accelForSpeed

        *

        imuSpeedPredictionTrust

        *

        IMU_ACCEL_TO_SPEED_GAIN

        *

        CONTROL_DT;

  
  

    // 单周期预测限幅

    if (

        imuDeltaSpeed

        >

        MAX_IMU_SPEED_DELTA_PER_CYCLE

    )

    {

        imuDeltaSpeed =

            MAX_IMU_SPEED_DELTA_PER_CYCLE;

    }

  
  

    if (

        imuDeltaSpeed

        <

        -MAX_IMU_SPEED_DELTA_PER_CYCLE

    )

    {

        imuDeltaSpeed =

            -MAX_IMU_SPEED_DELTA_PER_CYCLE;

    }

  
  

    vehicleSpeed +=

        imuDeltaSpeed;

  
  

    // --------------------------------------------------------

    // 2. 动态过程噪声Q

    //

    // 电机命令越大，系统越可能快速改变速度，

    // 因此允许Kalman状态更快变化。

    //

    // 注意：balanceOutput只调Q，不直接当作真实速度。

    // --------------------------------------------------------

  

    speedKalmanQ =

        SPEED_KALMAN_Q_BASE

        +

        SPEED_KALMAN_Q_MOTOR_GAIN

        *

        fabsf(balanceOutput);

  
  

    speedKalmanP +=

        speedKalmanQ

        *

        CONTROL_DT;

}

  
  

void correctVehicleSpeedKalman(

    float encoderMeasurement

)

{

    // --------------------------------------------------------

    // 编码器作为“速度绝对值主测量”

    //

    // 低速时虽然存在0/1 count量化，

    // 但R不再设得过大，避免Kalman长期跟着IMU预测跑偏。

    // --------------------------------------------------------

  

    if (

        fabsf(encoderMeasurement)

        <

        2.0f

    )

    {

        speedKalmanR =

            SPEED_KALMAN_R_LOW_SPEED;

    }

    else

    {

        speedKalmanR =

            SPEED_KALMAN_R_NORMAL;

    }

  
  

    float innovation =

        encoderMeasurement

        -

        vehicleSpeed;

  
  

    float innovationCovariance =

        speedKalmanP

        +

        speedKalmanR;

  
  

    float K =

        speedKalmanP

        /

        innovationCovariance;

  
  

    vehicleSpeed +=

        K

        *

        innovation;

  
  

    speedKalmanP =

        (1.0f - K)

        *

        speedKalmanP;

  
  

    // 防止P数值过小导致后续测量修正越来越迟钝

    if (

        speedKalmanP

        <

        0.02f

    )

    {

        speedKalmanP =

            0.02f;

    }

}

  
  

// ============================================================

// 漂移拉回外环

// 20Hz速度P + 300ms状态式漂移检测。

// ============================================================

  

void updateOuterLoop()

{

    int32_t count1;

    int32_t count2;

  

    noInterrupts();

  

    count1 =

        encoder1Count;

  

    count2 =

        encoder2Count;

  

    interrupts();

  
  

    // --------------------------------------------------------

    // 1. 50ms编码器增量

    // --------------------------------------------------------

  

    int32_t delta1 =

        count1

        -

        lastEncoder1Count;

  
  

    int32_t delta2 =

        count2

        -

        lastEncoder2Count;

  
  

    lastEncoder1Count =

        count1;

  

    lastEncoder2Count =

        count2;

  
  

    encoderSpeed1 =

        (float)delta1;

  
  

    encoderSpeed2 =

        (float)delta2;

  
  

    // --------------------------------------------------------

    // 2. 编码器车辆速度

    //

    // 当前沿用此前实测约定：

    // 向前 = 正

    // 向后 = 负

    // --------------------------------------------------------

  

    rawVehicleSpeed =

        -0.5f

        *

        (

            encoderSpeed1

            +

            encoderSpeed2

        );

  
  

    // --------------------------------------------------------

    // 3. 用编码器修正速度Kalman

    //

    // 此时vehicleSpeed已经由：

    // 400Hz IMU预测 + 20Hz编码器修正

    // 得到融合速度。

    // --------------------------------------------------------

  

    correctVehicleSpeedKalman(

        rawVehicleSpeed

    );

  
  

    // --------------------------------------------------------

    // 4. 限制目标速度

    // --------------------------------------------------------

  

    if (

        targetForwardSpeed

        >

        MAX_TARGET_FORWARD_SPEED

    )

    {

        targetForwardSpeed =

            MAX_TARGET_FORWARD_SPEED;

    }

  
  

    if (

        targetForwardSpeed

        <

        -MAX_TARGET_FORWARD_SPEED

    )

    {

        targetForwardSpeed =

            -MAX_TARGET_FORWARD_SPEED;

    }

  
  

    // --------------------------------------------------------

    // 5. 速度误差

    // --------------------------------------------------------

  

    speedError =

        targetForwardSpeed

        -

        vehicleSpeed;

  
  

    // --------------------------------------------------------

    // 6. 速度P直接STEP快速阻尼

    //

    // 不等待目标角 -> 车身倾斜 -> 直立环这一整段动态，

    // 而是根据融合速度直接给STEP一个小的反向修正。

    // --------------------------------------------------------

  

    speedDirectOutput =

        speedLoopDirection

        *

        speedDirectKp

        *

        speedError;

  
  

    if (

        speedDirectOutput

        >

        MAX_SPEED_DIRECT_OUTPUT

    )

    {

        speedDirectOutput =

            MAX_SPEED_DIRECT_OUTPUT;

    }

  
  

    if (

        speedDirectOutput

        <

        -MAX_SPEED_DIRECT_OUTPUT

    )

    {

        speedDirectOutput =

            -MAX_SPEED_DIRECT_OUTPUT;

    }

  
  

    // --------------------------------------------------------

    // 7. 速度积分

    //

    // 外环20Hz，因此dt = 0.05s。

    // 零附近不积分，防止编码器量化和IMU噪声造成慢性漂移。

    // --------------------------------------------------------

  

    if (

        fabsf(speedError)

        >

        SPEED_ERROR_DEADBAND

    )

    {

        speedIntegral +=

            speedError

            *

            0.05f;

    }

  
  

    // 积分限幅

    if (

        speedIntegral

        >

        SPEED_INTEGRAL_LIMIT

    )

    {

        speedIntegral =

            SPEED_INTEGRAL_LIMIT;

    }

  
  

    if (

        speedIntegral

        <

        -SPEED_INTEGRAL_LIMIT

    )

    {

        speedIntegral =

            -SPEED_INTEGRAL_LIMIT;

    }

  
  

    // --------------------------------------------------------

    // 8. 速度PI输出目标倾角

    // --------------------------------------------------------

  

    speedAngleCorrection =

        speedLoopDirection

        *

        (

            speedKp

            *

            speedError

            +

            speedKi

            *

            speedIntegral

        );

  
  

    // --------------------------------------------------------

    // 9. 外环目标倾角限幅

    // --------------------------------------------------------

  

    if (

        speedAngleCorrection

        >

        MAX_SPEED_ANGLE

    )

    {

        speedAngleCorrection =

            MAX_SPEED_ANGLE;

    }

  
  

    if (

        speedAngleCorrection

        <

        -MAX_SPEED_ANGLE

    )

    {

        speedAngleCorrection =

            -MAX_SPEED_ANGLE;

    }

  
  

    // --------------------------------------------------------

    // 10. 串级目标角

    // --------------------------------------------------------

  

    // targetBalanceAngle 不在20Hz外环里直接更新。

    // 由400Hz的 updateOuterLoopAuthority() 根据当前姿态平滑缩放，

    // 大角度时自动恢复到机械平衡角。

}

  
  

// ============================================================

// 外环权限管理

//

// 使用相对机械平衡角，而不是相对 targetBalanceAngle。

// 这样速度PI改变目标角时，不会影响“大角度判定”。

// ============================================================

  

void updateOuterLoopAuthority()

{

    float uprightError =

        pitch

        -

        balanceAngleOffset;

  

    float absUprightError =

        fabsf(

            uprightError

        );

  
  

    if (

        absUprightError

        <=

        OUTER_FULL_ANGLE_DEG

    )

    {

        outerLoopAuthority =

            1.0f;

    }

    else if (

        absUprightError

        <

        OUTER_DISABLE_ANGLE_DEG

    )

    {

        outerLoopAuthority =

            (

                OUTER_DISABLE_ANGLE_DEG

                -

                absUprightError

            )

            /

            (

                OUTER_DISABLE_ANGLE_DEG

                -

                OUTER_FULL_ANGLE_DEG

            );

    }

    else

    {

        outerLoopAuthority =

            0.0f;

    }

  
  

    // 目标角也必须受同一个权限控制。

    // 大角度时直接回到机械平衡角，让直立PD全力救车。

    targetBalanceAngle =

        balanceAngleOffset

        +

        outerLoopAuthority

        *

        speedAngleCorrection;

  
  

    // 大角度时释放历史积分，避免恢复后旧积分继续把车推走

    if (

        outerLoopAuthority

        <

        0.2f

    )

    {

        speedIntegral *=

            SPEED_INTEGRAL_DECAY_BIG_ANGLE;

    }

}

  
  

// ============================================================

// 位置快速拉回

//

// 400Hz直接读取编码器累计位置。

// 不经过50ms速度窗口，所以只要编码器出现少量位移就开始反向拉回。

// ============================================================

  

void updatePositionHold()

{

    int32_t count1;

    int32_t count2;

  

    noInterrupts();

  

    count1 =

        encoder1Count;

  

    count2 =

        encoder2Count;

  

    interrupts();

  
  

    // 与 rawVehicleSpeed 使用同一正负约定：

    // 向前 = 正，向后 = 负

    vehiclePosition =

        -0.5f

        *

        (

            (float)(

                count1

                -

                positionZeroCount1

            )

            +

            (float)(

                count2

                -

                positionZeroCount2

            )

        );

  
  

    float positionError =

        targetVehiclePosition

        -

        vehiclePosition;

  
  

    // 极小死区，避免编码器边沿抖动导致反复换向

    if (

        fabsf(positionError)

        <

        POSITION_DEADBAND_COUNT

    )

    {

        positionDirectOutput =

            0.0f;

  

        return;

    }

  
  

    positionDirectOutput =

        speedLoopDirection

        *

        positionDirectKp

        *

        positionError;

  
  

    if (

        positionDirectOutput

        >

        MAX_POSITION_DIRECT_OUTPUT

    )

    {

        positionDirectOutput =

            MAX_POSITION_DIRECT_OUTPUT;

    }

  
  

    if (

        positionDirectOutput

        <

        -MAX_POSITION_DIRECT_OUTPUT

    )

    {

        positionDirectOutput =

            -MAX_POSITION_DIRECT_OUTPUT;

    }

}

// 直立环 PD

// ============================================================

  

void updateBalanceLoop()

{

    float angleError =

        targetBalanceAngle

        -

        pitch;

  
  

    // --------------------------------------------------------

    // 1. 直立PD

    // --------------------------------------------------------

  

    balanceOutput =

        balanceDirection

        *

        (

            balanceKp

            *

            angleError

            -

            balanceKd

            *

            gyroPitchRate

        );

  
  

    // --------------------------------------------------------

    // 2. 直接速度/位置修正权限

    //

    // 与目标角使用同一个 outerLoopAuthority。

    // 大角度时速度环和位置环全部退出。

    // --------------------------------------------------------

  

    speedDirectAuthority =

        outerLoopAuthority;

  
  

    // --------------------------------------------------------

    // 3. 最终混合输出

    // --------------------------------------------------------

  

    finalBalanceOutput =

        balanceOutput

        +

        outerLoopAuthority

        *

        (

            speedDirectOutput

            +

            positionDirectOutput

        );

}

  
  

// ============================================================

// 转向环 PD

//

// 当前Kp/Kd = 0

// ============================================================

  

void updateTurnLoop()

{

    float turnError =

        targetTurnRate

        -

        gyroTurnRate;

  
  

    float turnDerivative =

        (

            turnError

            -

            lastTurnError

        )

        /

        CONTROL_DT;

  
  

    lastTurnError =

        turnError;

  
  

    turnOutput =

        turnKp

        *

        turnError

        +

        turnKd

        *

        turnDerivative;

}

// ============================================================

// loop

// 固定400Hz

// ============================================================

  

void loop()

{

    uint32_t nowUs =

        micros();

  
  

    // --------------------------------------------------------

    // 没到2.5ms就直接返回

    // --------------------------------------------------------

  

    if (

        (uint32_t)(

            nowUs

            -

            lastControlUs

        )

        <

        CONTROL_PERIOD_US

    )

    {

        return;

    }

  
  

    // --------------------------------------------------------

    // 保持固定时间基准

    // --------------------------------------------------------

  

    lastControlUs +=

        CONTROL_PERIOD_US;

  
  

    // --------------------------------------------------------

    // 读取BMI270

    // --------------------------------------------------------

  

    if (

        !readBMI270()

    )

    {

        imuFailCount++;

  

        if (

            imuFailCount

            <

            IMU_FAIL_CONFIRM_COUNT

        )

        {

            // 保持上一有效STEP输出

            return;

        }

  

        stopBothMotors();

  

        balanceOutput =

            0.0f;

  

        finalBalanceOutput =

            0.0f;

  

        speedDirectOutput =

            0.0f;

  

        speedDirectAuthority =

            0.0f;

  

        return;

    }

  

    imuFailCount =

        0;

  
  

    // --------------------------------------------------------

    // Kalman姿态

    // --------------------------------------------------------

  

    updateAttitude();

  
  

    // --------------------------------------------------------

    // 多源速度Kalman预测 400Hz

    //

    // IMU负责快速感知“开始加速”的趋势；

    // 编码器会在外环20Hz时进行绝对速度修正。

    // --------------------------------------------------------

  

    predictVehicleSpeedKalman();

  
  

    // --------------------------------------------------------

    // 蓝牙遥测20Hz

    // 放在倒地保护之前，这样即使触发倒地保护，

    // 手机上仍能继续看到IMU/Kalman数据。

    // --------------------------------------------------------

  

    telemetryDivider++;

  

    if (

        telemetryDivider

        >=

        TELEMETRY_DIVIDER

    )

    {

        telemetryDivider =

            0;

  

        sendTelemetry();

    }

  
  

    // --------------------------------------------------------

    // 倒地保护（连续确认，避免单帧尖峰突然停机）

    // --------------------------------------------------------

  

    float fallAngleError =

        fabsf(

            pitch

            -

            balanceAngleOffset

        );

  

    if (

        fallAngleError

        >=

        FALL_ANGLE_DEG

    )

    {

        fallConfirmCount++;

    }

    else

    {

        fallConfirmCount =

            0;

    }

  

    bool confirmedFall =

        fallConfirmCount

        >=

        FALL_CONFIRM_COUNT;

  

    bool emergencyFall =

        fallAngleError

        >=

        EMERGENCY_FALL_ANGLE_DEG;

  

    if (

        confirmedFall

        ||

        emergencyFall

    )

    {

        stopBothMotors();

  

        balanceOutput =

            0.0f;

  

        turnOutput =

            0.0f;

  

        speedError =

            0.0f;

  

        speedIntegral =

            0.0f;

  

        speedAngleCorrection =

            0.0f;

  

        speedDirectOutput =

            0.0f;

  

        speedDirectAuthority =

            0.0f;

  

        finalBalanceOutput =

            0.0f;

  

        outerLoopAuthority =

            0.0f;

  

        motor1ZeroHoldCount =

            0;

  

        motor2ZeroHoldCount =

            0;

  

        targetForwardSpeed =

            0.0f;

  

        noInterrupts();

  

        int32_t resetCount1 =

            encoder1Count;

  

        int32_t resetCount2 =

            encoder2Count;

  

        interrupts();

  

        lastEncoder1Count =

            resetCount1;

  

        lastEncoder2Count =

            resetCount2;

  

        positionZeroCount1 =

            resetCount1;

  

        positionZeroCount2 =

            resetCount2;

  

        vehiclePosition =

            0.0f;

  

        positionDirectOutput =

            0.0f;

  

        vehicleSpeed =

            0.0f;

  

        rawVehicleSpeed =

            0.0f;

  

        imuForwardAccelG =

            0.0f;

  

        imuForwardAccelFilteredG =

            0.0f;

  

        speedKalmanP =

            1.0f;

  

        targetBalanceAngle =

            balanceAngleOffset;

  

        return;

    }

  
  

    // --------------------------------------------------------

    // 速度PI外环20Hz，同时计算速度P直接STEP快速阻尼

    //

    // 400Hz控制循环，每20次执行一次 = 50ms。

    // 50ms编码器窗口用于降低低速时0/1计数跳变。

    // --------------------------------------------------------

  

    outerLoopDivider++;

  
  

    if (

        outerLoopDivider

        >=

        20

    )

    {

        outerLoopDivider =

            0;

  

        updateOuterLoop();

    }

  
  

    // --------------------------------------------------------

    // 外环权限管理400Hz

    // 大角度时速度/位置环平滑退出，直立PD优先

    // --------------------------------------------------------

  

    updateOuterLoopAuthority();

  
  

    // --------------------------------------------------------

    // 位置快速拉回400Hz

    // --------------------------------------------------------

  

    updatePositionHold();

  
  

    // --------------------------------------------------------

    // 直立环400Hz

    // --------------------------------------------------------
