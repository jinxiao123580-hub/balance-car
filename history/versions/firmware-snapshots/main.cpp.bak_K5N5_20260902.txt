// 蓝牙遥测

// 控制400Hz，每40个周期发送一次 = 10Hz

// 数据格式：D,ms,pitch100,gyro100,target100,balHz,spdAng100,auth1000,

// ============================================================
// 非阻塞遥测 txBuffer（9600 波特很慢，直接 print 会阻塞 400Hz 控制循环）
// 方案：sendTelemetry 只把整行写入环形缓冲；loop 末尾按 800B/s 预算 flush。
// 预算不足时丢整行（不部分发送）。
// ============================================================

static const uint16_t TX_BUF_SIZE = 768;

static char txBuffer[TX_BUF_SIZE];

static volatile uint16_t txHead = 0;

static volatile uint16_t txTail = 0;

// 每控制周期(2.5ms)累加的发送预算（字节）
// 800B/s * 0.0025s = 2B/周期
static const float TX_BUDGET_PER_CYCLE = 2.0f;

static float txBudgetBytes = 0.0f;

static const float TX_BUDGET_MAX = 40.0f;   // 预算上限，避免积压暴写

// 写入环形缓冲（整行入队；空间不足则丢弃该行）
static bool txPushLine(const char* line, uint16_t len)
{
    if (len == 0 || len >= TX_BUF_SIZE)
        return false;
    // 检查是否有连续空间放 [len 字节 + 换行]
    uint16_t space;
    if (txHead >= txTail)
        space = (TX_BUF_SIZE - 1) - (txHead - txTail);
    else
        space = (txTail - txHead) - 1;
    if (space < len + 1)
        return false;
    for (uint16_t i = 0; i < len; i++)
    {
        txBuffer[txHead] = line[i];
        txHead = (txHead + 1) % TX_BUF_SIZE;
    }
    txBuffer[txHead] = '\n';
    txHead = (txHead + 1) % TX_BUF_SIZE;
    return true;
}

// loop 末尾按预算 flush（非阻塞：只写 UART 当前可容纳的字节）
void flushTelemetryTx()
{
    txBudgetBytes += TX_BUDGET_PER_CYCLE;
    if (txBudgetBytes > TX_BUDGET_MAX)
        txBudgetBytes = TX_BUDGET_MAX;

    while (txHead != txTail && txBudgetBytes >= 1.0f)
    {
        int avail = BTSerial.availableForWrite();
        if (avail <= 0)
            break;
        int toSend = (int)txBudgetBytes;
        if (toSend > avail)
            toSend = avail;
        if (toSend <= 0)
            break;
        uint16_t n = 0;
        while (n < toSend && txHead != txTail)
        {
            BTSerial.write((uint8_t)txBuffer[txTail]);
            txTail = (txTail + 1) % TX_BUF_SIZE;
            n++;
        }
        txBudgetBytes -= (float)n;
    }
}

// MPU6050底层寄存器写（函数名沿用BMI270，避免改动所有调用点）

// ============================================================

  

bool bmiWriteRegister(

    uint8_t reg,

    uint8_t value

)

{

    BMIWire.beginTransmission(

        MPU6050_ADDRESS

    );

  

    BMIWire.write(reg);

    BMIWire.write(value);

  

    uint8_t error =

        BMIWire.endTransmission();

  

    delayMicroseconds(500);

  

    return

        error == 0;

}

  
  

// ============================================================

// MPU6050底层寄存器读（函数名沿用BMI270）

// ============================================================

  

bool bmiReadRegister(

    uint8_t reg,

    uint8_t &value

)

{

    BMIWire.beginTransmission(

        MPU6050_ADDRESS

    );

  

    BMIWire.write(reg);

  

    if (

        BMIWire.endTransmission(false)

        != 0

    )

    {

        return false;

    }

  
  

    uint8_t count =

        BMIWire.requestFrom(

            MPU6050_ADDRESS,

            (uint8_t)1

        );

  
  

    if (

        count != 1

        ||

        !BMIWire.available()

    )

    {

        return false;

    }

  
  

    value =

        BMIWire.read();

  
  

    return true;

}

  
  

// ============================================================

// 配置MPU6050为400Hz（函数名沿用BMI270）

// ============================================================

  

bool configureBMI270_400Hz()
{
    // ---- MPU6050 配置（沿用原函数名，setup()调用点不变）----
    // 1. 采样率分频：8kHz/(19+1)=400Hz
    if (!bmiWriteRegister(MPU6050_REG_SMPLRT_DIV, MPU6050_SMPLRT_DIV_400HZ))
    {
        return false;
    }
    delay(5);

    // 2. DLPF 94Hz（增强滤波：MPU6050噪声大于BMI270，184Hz→94Hz）
    if (!bmiWriteRegister(MPU6050_REG_CONFIG, MPU6050_DLPF_BW_94HZ))
    {
        return false;
    }
    delay(5);

    // 3. 陀螺 ±250dps
    if (!bmiWriteRegister(MPU6050_REG_GYRO_CONFIG, MPU6050_GYRO_RANGE_250))
    {
        return false;
    }
    delay(5);

    // 4. 加速度 ±2g
    if (!bmiWriteRegister(MPU6050_REG_ACCEL_CONFIG, MPU6050_ACCEL_RANGE_2G))
    {
        return false;
    }
    delay(5);

    // 5. 唤醒（清除SLEEP位）
    if (!bmiWriteRegister(MPU6050_REG_PWR_MGMT_1, 0x00))
    {
        return false;
    }
    delay(10);

    // 回读确认
    uint8_t val = 0;
    if (!bmiReadRegister(MPU6050_REG_GYRO_CONFIG, val)) { return false; }
    if (val != MPU6050_GYRO_RANGE_250) { return false; }

    return true;
}
  
  

// ============================================================

// MPU6050读取（函数名沿用BMI270）

// ============================================================

  

bool readBMI270()
{
    // MPU6050：读 0x3B 起 14 字节（ACCEL XYZ + TEMP + GYRO XYZ，int16 大端）
    BMIWire.beginTransmission(MPU6050_ADDRESS);
    BMIWire.write(MPU6050_REG_ACCEL_XOUT_H);
    if (BMIWire.endTransmission(false) != 0) { return false; }

    if (BMIWire.requestFrom((int)MPU6050_ADDRESS, 14) != 14)
    {
        return false;
    }

    int16_t aX = (int16_t)((BMIWire.read() << 8) | BMIWire.read());
    int16_t aY = (int16_t)((BMIWire.read() << 8) | BMIWire.read());
    int16_t aZ = (int16_t)((BMIWire.read() << 8) | BMIWire.read());
    /* temp */ BMIWire.read(); BMIWire.read();
    int16_t gX = (int16_t)((BMIWire.read() << 8) | BMIWire.read());
    int16_t gY = (int16_t)((BMIWire.read() << 8) | BMIWire.read());
    int16_t gZ = (int16_t)((BMIWire.read() << 8) | BMIWire.read());

    // LSB → g / dps
    ax = (float)aX * MPU6050_ACCEL_SCALE;
    ay = (float)aY * MPU6050_ACCEL_SCALE;
    az = (float)aZ * MPU6050_ACCEL_SCALE;
    gx = (float)gX * MPU6050_GYRO_SCALE;
    gy = (float)gY * MPU6050_GYRO_SCALE;
    gz = (float)gZ * MPU6050_GYRO_SCALE;

    return true;
}
  
  

// ============================================================

// MPU6050初始化（函数名沿用BMI270）

// ============================================================

  

bool initBMI270()
{
    delay(1000);

    BMIWire.begin();
    BMIWire.setClock(400000);
    delay(200);

    for (int i = 0; i < 10; i++)
    {
        // 读 WHO_AM_I，应回 0x68
        uint8_t whoAmI = 0;
        if (bmiReadRegister(MPU6050_REG_WHO_AM_I, whoAmI) && whoAmI == 0x68)
        {
            delay(200);

            // 唤醒 + 重置配置（清SLEEP）
            if (bmiWriteRegister(MPU6050_REG_PWR_MGMT_1, 0x00))
            {
                delay(100);
                return true;
            }
        }

        delay(300);
    }

    return false;
}
  
  

// ============================================================

// 加速度Pitch

// X = 上下

// Z = 前后

// ============================================================

  

float calculateAccelPitch()

{

    return

        atan2f(

            -az,

            -ax

        )

        *

        180.0f

        /

        PI;

}

  
  

// ============================================================

// IMU校准

// 开机保持车体静止

// ============================================================

  

bool calibrateIMU()
{
    // 车开机任意姿态，静止待 It 到重力方向稳定后，把重力竖直锚定为0°。
    // pitchZero = 静止时 calculateAccelPitch() 的均值 = 重力在IMU坐标系的方向角。
    // 因此 0° 恒等于重力法向（绝对竖直），完全自适应IMU安装方向，
    // 拨动IMU后下次开机自动重新对准 —— 不写死任何安装角。
    const float CAL_STATIC_GYRO_DPS = 4.0f;   // 静止判据：角速度小于此值(°/s)
    const float CAL_GRAV_MAG_TOL = 0.25f;      // 静止判据：|模长-1g| 容差(g)
    const int SAMPLE_TARGET = 400;              // 需累计的有效样本数
    const uint32_t CAL_TIMEOUT_MS = 20000;      // 总等待上限(毫秒) 保护

    delay(300);   // 上电稳定
    float pitchSum = 0.0f, gyroYSum = 0.0f, gyroXSum = 0.0f;
    int valid = 0;
    uint32_t tStart = millis();
    while (valid < SAMPLE_TARGET && (millis() - tStart) < CAL_TIMEOUT_MS)
    {
        if (readBMI270())
        {
            float mag = sqrtf(ax*ax + ay*ay + az*az);
            bool still = (fabsf(gy) < CAL_STATIC_GYRO_DPS)
                      && (fabsf(mag - 1.0f) < CAL_GRAV_MAG_TOL);
            if (still)
            {
                pitchSum += calculateAccelPitch();
                gyroYSum += gy;
                gyroXSum += gx;
                valid++;
            }
        }
        delay(3);
    }
    if (valid < SAMPLE_TARGET / 2)
    {
        return false;
    }
    pitchZero = pitchSum / (float)valid;   // 重力方向绝对角 -> 0°基准
    gyroYZero = gyroYSum / (float)valid;   // 陀螺零偏
    gyroXZero = gyroXSum / (float)valid;
    accPitch  = 0.0f;
    pitch     = 0.0f;
    kalmanAngle = 0.0f;
    kalmanBias  = 0.0f;
    P00 = 1.0f; P01 = 0.0f; P10 = 0.0f; P11 = 1.0f;
    return true;
}

  
  

// ============================================================

// Kalman

// ============================================================

  

float updateKalman(

    float measuredAngle,

    float measuredRate,

    float dt

)

{

    // --------------------------------------------------------

    // 1. 状态预测

    // --------------------------------------------------------

  

    float rate =

        measuredRate

        -

        kalmanBias;

  
  

    kalmanAngle +=

        dt

        *

        rate;

  
  

    // --------------------------------------------------------

    // 2. 协方差预测

    // --------------------------------------------------------

  

    P00 +=

        dt

        *

        (

            dt * P11

            -

            P01

            -

            P10

            +

            Q_angle

        );

  
  

    P01 -=

        dt

        *

        P11;

  
  

    P10 -=

        dt

        *

        P11;

  
  

    P11 +=

        Q_bias

        *

        dt;

  
  

    // --------------------------------------------------------

    // 3. 测量创新

    // --------------------------------------------------------

  

    float innovation =

        measuredAngle

        -

        kalmanAngle;

  
  

    // --------------------------------------------------------

    // 动态测量噪声

    //

    // 车辆静止/低动态时 dynamicR ≈ 0.03

    // 车辆加速或振动明显时 dynamicR 增大，

    // 从而暂时减少加速度角对Kalman的修正。

    // --------------------------------------------------------

  

    float S =

        P00

        +

        dynamicR;

  
  

    // --------------------------------------------------------

    // 4. Kalman增益

    // --------------------------------------------------------

  

    float K0 =

        P00

        /

        S;

  
  

    float K1 =

        P10

        /

        S;

  
  

    // --------------------------------------------------------

    // 5. 状态修正

    // --------------------------------------------------------

  

    kalmanAngle +=

        K0

        *

        innovation;

  
  

    kalmanBias +=

        K1

        *

        innovation;

  
  

    // --------------------------------------------------------

    // 6. 协方差修正

    // --------------------------------------------------------

  

    float oldP00 = P00;

    float oldP01 = P01;

  
  

    P00 -=

        K0

        *

        oldP00;

  
  

    P01 -=

        K0

        *

        oldP01;

  
  

    P10 -=

        K1

        *

        oldP00;

  
  

    P11 -=

        K1

        *

        oldP01;

  
  

    return

        kalmanAngle;

}

  
  

// ============================================================

// 姿态更新

// ============================================================

  

void updateAttitude()

{

    // --------------------------------------------------------

    // 1. 原始加速度角

    // --------------------------------------------------------

  

    accPitch =

        calculateAccelPitch()

        -

        pitchZero;

  
  

    // --------------------------------------------------------

    // 2. Pitch轴陀螺仪角速度

    // --------------------------------------------------------

  

    gyroPitchRate =

        GY_DIRECTION

        *

        (

            gy

            -

            gyroYZero

        );

  
  

    // --------------------------------------------------------

    // 3. 计算三轴加速度模长

    //

    // 加速度量按g量级使用（MPU6050 ±2g 量程 /16384 换算）。

    // 静止时理论上接近1g。

    // --------------------------------------------------------

  

    accelMagnitude =

        sqrtf(

            ax * ax

            +

            ay * ay

            +

            az * az

        );

  
  

    accelMagnitudeError =

        fabsf(

            accelMagnitude

            -

            1.0f

        );

  
  

    // --------------------------------------------------------

    // 4. 根据 |a|-1g 动态调整R

    //

    // 越偏离1g，说明线性加速度/振动越明显，

    // 这时accPitch可信度降低。

    // --------------------------------------------------------

  

    // 连续插值：加速度模长误差 0 -> ACC_ERR_HIGH 平滑映射 R
    // 静止/低速：R小(0.02)，acc权重高，抑制陀螺零偏积分漂移
    // 高速/振动：R大(1.20)，acc权重低，防止线性加速度污染角度
    float errRatio = accelMagnitudeError / ACC_ERR_HIGH;
    if (errRatio > 1.0f) { errRatio = 1.0f; }
    dynamicR = R_LOW_DYNAMIC + (R_HIGH_DYNAMIC - R_LOW_DYNAMIC) * errRatio;

  
  

    // --------------------------------------------------------

    // 5. Kalman融合

    // --------------------------------------------------------

  

    pitch =

        updateKalman(

            accPitch,

            gyroPitchRate,

            CONTROL_DT

        );

  
  

    // --------------------------------------------------------

    // 6. 根据Pitch去除Z轴重力分量，得到前向线加速度趋势

    //

    // 静止时：

    // az ≈ -sin(pitch)

    //

    // 因此：

    // az + sin(pitch) ≈ 0

    //

    // 再乘方向参数统一为“向前加速为正”。

    // --------------------------------------------------------

  

    float pitchRad =

        pitch

        *

        DEG_TO_RAD;

  
  

    imuForwardAccelG =

        IMU_FORWARD_ACCEL_DIRECTION

        *

        (

            az

            +

            sinf(pitchRad)

        );

  
  

    // 尖峰限幅

    if (

        imuForwardAccelG

        >

        IMU_ACCEL_CLAMP_G

    )

    {

        imuForwardAccelG =

            IMU_ACCEL_CLAMP_G;

    }

  
  

    if (

        imuForwardAccelG

        <

        -IMU_ACCEL_CLAMP_G

    )

    {

        imuForwardAccelG =

            -IMU_ACCEL_CLAMP_G;

    }

  
  

    // 轻度低通

    imuForwardAccelFilteredG =

        (1.0f - IMU_ACCEL_LPF_ALPHA)

        *

        imuForwardAccelFilteredG

        +

        IMU_ACCEL_LPF_ALPHA

        *

        imuForwardAccelG;

  
  

    // 根据加速度模长偏离1g的程度降低IMU预测可信度

    if (

        accelMagnitudeError

        <

        ACC_ERR_LOW

    )

    {

        imuSpeedPredictionTrust =

            1.0f;

    }

    else if (

        accelMagnitudeError

        <

        ACC_ERR_HIGH

    )

    {

        imuSpeedPredictionTrust =

            0.45f;

    }

    else

    {

        imuSpeedPredictionTrust =

            0.12f;

    }

  
  

    // --------------------------------------------------------

    // 7. 转向角速度

    // --------------------------------------------------------

  

    gyroTurnRate =

        TURN_GYRO_DIRECTION

        *

        (

            gx

            -

            gyroXZero

        );

}

)

{

    if (

        !isfinite(commandHz)

    )

    {

        return;

    }

  

    if (

        fabsf(commandHz)

        <

        MOTOR_STOP_THRESHOLD_HZ

    )

    {

        if (

            HOLD_LAST_OUTPUT_ON_ZERO

            &&

            motor1ZeroHoldCount

            <

            ZERO_HOLD_MAX_CYCLES

        )

        {

            motor1ZeroHoldCount++;

            return;

        }

  

        motor1ZeroHoldCount =

            0;

  

        stopMotor1();

        return;

    }

  

    // 新有效输出到来，清零0输出保持计数

    motor1ZeroHoldCount =

        0;

  
  

    bool dir =

        commandHz

        >=

        0.0f;

  
  

    if (

        motor1Invert

    )

    {

        dir =

            !dir;

    }

  
  

    bool directionChanged =

        (

            motor1LastCommandHz

            *

            commandHz

        )

        <

        0.0f;

  
  

    if (

        directionChanged

    )

    {

        motor1Timer.pause();

    }

  
  

    digitalWrite(

        MOTOR1_DIR_PIN,

        dir ? HIGH : LOW

    );

  
  

    if (

        directionChanged

    )

    {

        delayMicroseconds(2);

    }

  
  

    float frequency =

        fabsf(

            commandHz

        );

  
  

    bool frequencyChanged =

        motor1LastCommandHz

        ==

        0.0f

        ||

        fabsf(

            fabsf(

                motor1LastCommandHz

            )

            -

            frequency

        )

        >=

        PWM_UPDATE_MIN_DELTA_HZ;

  
  

    if (

        directionChanged

        ||

        frequencyChanged

    )

    {

        motor1Timer.setPWM(

            1,

            MOTOR1_STEP_PIN,

            (uint32_t)frequency,

            50

        );

  
  

        motor1Timer.resume();

  
  

        motor1LastCommandHz =

            commandHz;

    }

}

// ============================================================

// 多源速度Kalman

//

// 400Hz：用IMU前向线加速度做预测

// 20Hz ：用编码器速度做测量修正

//

// 电机命令不直接当“真实速度”，只用来动态调整过程噪声Q：

// 控制命令大 -> 系统更可能快速改变速度 -> Kalman更允许状态变化。

// ============================================================

  

void predictVehicleSpeedKalman()

{

    // --------------------------------------------------------

    // 1. IMU只负责“短期趋势预测”

    //

    // 编码器负责长期速度绝对值；

    // IMU不允许长期主导vehicleSpeed。

    // --------------------------------------------------------

  

    float accelForSpeed =

        imuForwardAccelFilteredG;

  
  

    // 小加速度死区：

    // 防止0.01~0.02g级别的零偏不断积分成假速度。

    if (

        fabsf(accelForSpeed)

        <

        IMU_SPEED_ACCEL_DEADBAND_G

    )

    {

        accelForSpeed =

            0.0f;

    }

  
  

    float imuDeltaSpeed =

        accelForSpeed

        *

        imuSpeedPredictionTrust

        *

        IMU_ACCEL_TO_SPEED_GAIN

        *

        CONTROL_DT;

  
  

    // 单周期预测限幅

    if (

        imuDeltaSpeed

        >

        MAX_IMU_SPEED_DELTA_PER_CYCLE

    )

    {

        imuDeltaSpeed =

            MAX_IMU_SPEED_DELTA_PER_CYCLE;

    }

  
  

    if (

        imuDeltaSpeed

        <

        -MAX_IMU_SPEED_DELTA_PER_CYCLE

    )

    {

        imuDeltaSpeed =

            -MAX_IMU_SPEED_DELTA_PER_CYCLE;

    }

  
  

    vehicleSpeed +=

        imuDeltaSpeed;

  
  

    // --------------------------------------------------------

    // 2. 动态过程噪声Q

    //

    // 电机命令越大，系统越可能快速改变速度，

    // 因此允许Kalman状态更快变化。

    //

    // 注意：balanceOutput只调Q，不直接当作真实速度。

    // --------------------------------------------------------

  

    speedKalmanQ =

        SPEED_KALMAN_Q_BASE

        +

        SPEED_KALMAN_Q_MOTOR_GAIN

        *

        fabsf(balanceOutput);

  
  

    speedKalmanP +=

        speedKalmanQ

        *

        CONTROL_DT;

}

  
  

void correctVehicleSpeedKalman(

    float encoderMeasurement

)

{

    // --------------------------------------------------------

    // 编码器作为“速度绝对值主测量”

    //

    // 低速时虽然存在0/1 count量化，

    // 但R不再设得过大，避免Kalman长期跟着IMU预测跑偏。

    // --------------------------------------------------------

  

    if (

        fabsf(encoderMeasurement)

        <

        2.0f

    )

    {

        speedKalmanR =

            SPEED_KALMAN_R_LOW_SPEED;

    }

    else

    {

        speedKalmanR =

            SPEED_KALMAN_R_NORMAL;

    }

  
  

    float innovation =

        encoderMeasurement

        -

        vehicleSpeed;

  
  

    float innovationCovariance =

        speedKalmanP

        +

        speedKalmanR;

  
  

    float K =

        speedKalmanP

        /

        innovationCovariance;

  
  

    vehicleSpeed +=

        K

        *

        innovation;

  
  

    speedKalmanP =

        (1.0f - K)

        *

        speedKalmanP;

  
  

    // 防止P数值过小导致后续测量修正越来越迟钝

    if (

        speedKalmanP

        <

        0.02f

    )

    {

        speedKalmanP =

            0.02f;

    }

}

  
  

// ============================================================

// 漂移拉回外环

// 20Hz速度P + 300ms状态式漂移检测。

// ============================================================

  

void updateOuterLoop()

{

    int32_t count1;

    int32_t count2;

  

    noInterrupts();

  

    count1 =

        encoder1Count;

  

    count2 =

        encoder2Count;

  

    interrupts();

  
  

    // --------------------------------------------------------

    // 1. 50ms编码器增量

    // --------------------------------------------------------

  

    int32_t delta1 =

        count1

        -

        lastEncoder1Count;

  
  

    int32_t delta2 =

        count2

        -

        lastEncoder2Count;

  
  

    lastEncoder1Count =

        count1;

  

    lastEncoder2Count =

        count2;

  
  

    encoderSpeed1 =

        (1.0f - ENCODER_SPEED_LPF_ALPHA) * encoderSpeed1 + ENCODER_SPEED_LPF_ALPHA * (float)delta1;

  
  

    encoderSpeed2 =

        (1.0f - ENCODER_SPEED_LPF_ALPHA) * encoderSpeed2 + ENCODER_SPEED_LPF_ALPHA * (float)delta2;

  
  

    // --------------------------------------------------------

    // 2. 编码器车辆速度

    //

    // 当前沿用此前实测约定：

    // 向前 = 正

    // 向后 = 负

    // --------------------------------------------------------

  

    rawVehicleSpeed =

        -0.5f

        *

        (

            encoderSpeed1

            +

            encoderSpeed2

        );

  
  

    // --------------------------------------------------------

    // 3. 用编码器修正速度Kalman

    //

    // 此时vehicleSpeed已经由：

    // 400Hz IMU预测 + 20Hz编码器修正

    // 得到融合速度。

    // --------------------------------------------------------

  

    correctVehicleSpeedKalman(

        rawVehicleSpeed

    );

  
  

    // --------------------------------------------------------

    // 4. 限制目标速度

    // --------------------------------------------------------

  

    if (

        targetForwardSpeed

        >

        MAX_TARGET_FORWARD_SPEED

    )

    {

        targetForwardSpeed =

            MAX_TARGET_FORWARD_SPEED;

    }

  
  

    if (

        targetForwardSpeed

        <

        -MAX_TARGET_FORWARD_SPEED

    )

    {

        targetForwardSpeed =

            -MAX_TARGET_FORWARD_SPEED;

    }

  
  

    // --------------------------------------------------------

    // 5. 速度误差

    // --------------------------------------------------------

  

    speedError =

        targetForwardSpeed

        -

        vehicleSpeed;

  
  

    // --------------------------------------------------------

    // 6. 速度P直接STEP快速阻尼

    //

    // 不等待目标角 -> 车身倾斜 -> 直立环这一整段动态，

    // 而是根据融合速度直接给STEP一个小的反向修正。

    // --------------------------------------------------------

  

    // 直投积分累积（20Hz外环 dt=0.05）：对准 speedError，治持续单向溜车。
    // 只在误差超死区累积（防噪声漂移），带限幅防积分过冲。
    if (fabsf(speedError) > SPEED_ERROR_DEADBAND)
    {
        speedIntegral += speedError * 0.05f;
        if (speedIntegral > SPEED_INTEGRAL_LIMIT) speedIntegral = SPEED_INTEGRAL_LIMIT;
        else if (speedIntegral < -SPEED_INTEGRAL_LIMIT) speedIntegral = -SPEED_INTEGRAL_LIMIT;
    }

    speedAngleCorrection = speedLoopDirection * (speedKp * speedError + speedKi * speedIntegral);

  
  

    if (speedAngleCorrection > MAX_SPEED_ANGLE)
            speedAngleCorrection = MAX_SPEED_ANGLE;

  
  

    if (speedAngleCorrection < -MAX_SPEED_ANGLE)
            speedAngleCorrection = -MAX_SPEED_ANGLE;

  
  

    // --------------------------------------------------------

}

  
  

// ============================================================

// 外环权限管理

//

// 使用相对机械平衡角，而不是相对 targetBalanceAngle。

// 这样速度PI改变目标角时，不会影响“大角度判定”。

// ============================================================

  

void updateOuterLoopAuthority()

{

    float uprightError =

        pitch

        -

        balanceAngleOffset;

  

    float absUprightError =

        fabsf(

            uprightError

        );

  
  

    if (

        absUprightError

        <=

        OUTER_FULL_ANGLE_DEG

    )

    {

        outerLoopAuthority =

            1.0f;

    }

    else if (

        absUprightError

        <

        OUTER_DISABLE_ANGLE_DEG

    )

    {

        outerLoopAuthority =

            (

                OUTER_DISABLE_ANGLE_DEG

                -

                absUprightError

            )

            /

            (

                OUTER_DISABLE_ANGLE_DEG

                -

                OUTER_FULL_ANGLE_DEG

            );

    }

    else

    {

        outerLoopAuthority =

            0.0f;

    }

  
  

    // 目标角必须受同一个权限控制。

    // 大角度时直接回到机械平衡角，让直立PD全力救车。

    // 用户方案：速度环不再改目标角（只直接叠加轮速 speedDirect），

    // 因此这里目标角固定 = 机械平衡角；速度PI把 speedAngleCorrection 叠加到目标角，让车倾斜以移动。

    targetBalanceAngle =

        balanceAngleOffset

        +

        outerLoopAuthority

        *

        speedAngleCorrection;

  
  

    // 大角度时释放历史积分，避免恢复后旧积分继续把车推走

    if (

        outerLoopAuthority

        <

        0.2f

    )

    {

        speedIntegral *= SPEED_INTEGRAL_DECAY_BIG_ANGLE;

    }

}

  
  

// ============================================================

// 位置快速拉回

//

// 400Hz直接读取编码器累计位置。

// 不经过50ms速度窗口，所以只要编码器出现少量位移就开始反向拉回。

// ============================================================

  

void updatePositionHold()

{

    int32_t count1;

    int32_t count2;

  

    noInterrupts();

  

    count1 =

        encoder1Count;

  

    count2 =

        encoder2Count;

  

    interrupts();

  
  

    // 与 rawVehicleSpeed 使用同一正负约定：

    // 向前 = 正，向后 = 负

    vehiclePosition =

        -0.5f

        *

        (

            (float)(

                count1

                -

                positionZeroCount1

            )

            +

            (float)(

                count2

                -

                positionZeroCount2

            )

        );

  
  

    float positionError =

        targetVehiclePosition

        -

        vehiclePosition;

  
  

    // 极小死区，避免编码器边沿抖动导致反复换向

    if (

        fabsf(positionError)

        <

        POSITION_DEADBAND_COUNT

    )

    {

        positionDirectOutput =

            0.0f;

  

        return;

    }

  
  

    positionDirectOutput =

        speedLoopDirection

        *

        positionDirectKp

        *

        positionError;

  
  

    if (

        positionDirectOutput

        >

        MAX_POSITION_DIRECT_OUTPUT

    )

    {

        positionDirectOutput =

            MAX_POSITION_DIRECT_OUTPUT;

    }

  
  

    if (

        positionDirectOutput

        <

        -MAX_POSITION_DIRECT_OUTPUT

    )

    {

        positionDirectOutput =

            -MAX_POSITION_DIRECT_OUTPUT;

    }

}

  
  

// ============================================================

// 直立环 PD

// ============================================================

  

void updateBalanceLoop()

{

    float angleError =

        targetBalanceAngle

        -

        pitch;

  
  

    // --------------------------------------------------------

    // 1. 直立PD

    // --------------------------------------------------------

    // 角度死区：|pitch - target| < BALANCE_ANGLE_DEADBAND_DEG 时直立输出0

    if (fabsf(angleError) < BALANCE_ANGLE_DEADBAND_DEG)

    {

        balanceOutput = 0.0f;

    }

    else

    {

    balanceOutput =

        balanceDirection

        *

        (

            balanceKp

            *

            angleError

            -

            balanceKd

            *

            gyroPitchRate

        );

    }

  
  

    // --------------------------------------------------------

    // 2. 直接速度/位置修正权限

    //

    // 与目标角使用同一个 outerLoopAuthority。

    // 大角度时速度环和位置环全部退出。

    // --------------------------------------------------------

  

    speedDirectAuthority =

        outerLoopAuthority;

  
  

    // --------------------------------------------------------

    // 3. 最终混合输出

    // --------------------------------------------------------

  

    finalBalanceOutput =

        balanceOutput;

    // 8/21 输出限幅：无此限幅时大角度下 finalBalanceOutput 可达 ±25kHz，
    // DRV8825 超高频失步/转矩骤降 → "偏了很难拉回来"。限幅到步进电机有效工作区。
    if (finalBalanceOutput > BALANCE_OUTPUT_LIMIT_HZ)

    {

        finalBalanceOutput = BALANCE_OUTPUT_LIMIT_HZ;

    }

    if (finalBalanceOutput < -BALANCE_OUTPUT_LIMIT_HZ)

    {

        finalBalanceOutput = -BALANCE_OUTPUT_LIMIT_HZ;

    }

    // 8/21 输出斜坡限速：400Hz控制下每周期(2.5ms)最多变化±BALANCE_OUTPUT_SLEW_HZ。
    // 修复"运行中突然卡死"：无斜坡时大角度瞬间从0跳到±25kHz，
    // 步进电机加速跟不上→失步→转子卡在齿槽位置→抱死（用户实测确认）。
    // 斜坡让频率平滑爬升，电机能跟上；同时保留大角度恢复力（限幅12000不砍出力）。
    float slewDelta = finalBalanceOutput - lastFinalBalanceOutput;
    if (slewDelta > BALANCE_OUTPUT_SLEW_HZ)

    {

        finalBalanceOutput = lastFinalBalanceOutput + BALANCE_OUTPUT_SLEW_HZ;

    }
    else if (slewDelta < -BALANCE_OUTPUT_SLEW_HZ)

    {

        finalBalanceOutput = lastFinalBalanceOutput - BALANCE_OUTPUT_SLEW_HZ;

    }
    lastFinalBalanceOutput = finalBalanceOutput;

}

  
  

// ============================================================

// 转向环 PD

//

// 当前Kp/Kd = 0

// ============================================================

  

void updateTurnLoop()

{

    float turnError =

        targetTurnRate

        -

        gyroTurnRate;

  
  

    float turnDerivative =

        (

            turnError

            -

            lastTurnError

        )

        /

        CONTROL_DT;

  
  

    lastTurnError =

        turnError;

  
  

    turnOutput =

        turnKp

        *

        turnError

        +

        turnKd

        *

        turnDerivative;

}

  
  

// ============================================================

// 蓝牙发送

// 网页原来的解析格式无需修改：

// time_ms,accPitch,kalmanPitch

// ============================================================

  

void sendTelemetry()
{
    // 18 列诊断快照，全部整数缩放以节省 9600 波特带宽
    // D,ms,pitch100,gyro100,target100,balHz,spdAng100,auth1000,
    //   spdDirHz,posDirHz,finalHz,speed1000,raw1000,pos100,
    //   enc1,enc2,m1Hz,m2Hz
    char line[192];
    int n = snprintf(line, sizeof(line),
        "D,%lu,%d,%d,%d,%d,%d,%d,%d,%d,%d,%d,%d,%d,%ld,%ld,%d,%d",
        (unsigned long)millis(),
        (int)(pitch * 100.0f),                 // 1  Kalman倾角 *100
        (int)(gyroPitchRate * 100.0f),         // 2  角速度 *100
        (int)(targetBalanceAngle * 100.0f),    // 3  目标角 *100
        (int)balanceOutput,                    // 4  直立PD输出 Hz
        (int)(speedAngleCorrection * 100.0f),     // 5  速度PI目标倾角修正 *100（经典串级）
        (int)(outerLoopAuthority * 1000.0f),   // 6  外环权限 *1000
        (int)(speedDirectOutput * speedDirectAuthority), // 7 速度直接STEP Hz
        (int)positionDirectOutput,             // 8  位置修正 Hz
        (int)finalBalanceOutput,               // 9  最终输出 Hz
        (int)(vehicleSpeed * 1000.0f),         // 10 融合速度 *1000
        (int)(rawVehicleSpeed * 1000.0f),      // 11 编码器原始速度 *1000
        (int)(vehiclePosition * 100.0f),       // 12 位置 *100
        (long)encoder1Count,                   // 13 编码器1计数
        (long)encoder2Count,                   // 14 编码器2计数
        (int)motor1LastCommandHz,              // 15 电机1实际命令 Hz
        (int)motor2LastCommandHz);             // 16 电机2实际命令 Hz
    if (n > 0)
        txPushLine(line, (uint16_t)n);
}

        0;

  

    motor2ZeroHoldCount =

        0;

  

    outerLoopAuthority =

        1.0f;

  

    // 网页会忽略这一行表头

    BTSerial.println(

        "time_ms,fused_speed,speed_direct_hz"

    );

  
  

    lastControlUs =

        micros();

}

// ============================================================
// 蓝牙在线调参（在 loop 每圈开头调用，不影响400Hz控制时序）
// 命令格式：一行一个，回车发送。 字母 + 空格 + 数值
//   P <kp>    改 balanceKp
//   D <kd>    改 balanceKd
//   O <offset> 改 balanceAngleOffset（平衡点）
//   Z <deg>   改 BALANCE_ANGLE_DEADBAND_DEG（死区）
//   L <hz>    改 BALANCE_OUTPUT_LIMIT_HZ（输出限幅）
//   S <hz>    改 BALANCE_OUTPUT_SLEW_HZ（每周期最大变化）
//   ?         回显当前全部参数
// ============================================================

static char btCmdBuf[48];
static uint8_t btCmdLen = 0;

void handleBtCommand(char* cmd, uint8_t len);   // 前置声明

void pollBtCommand()
{
    while (BTSerial.available() > 0) {
        char c = (char)BTSerial.read();
        if (c == '\n' || c == '\r') {
            handleBtCommand(btCmdBuf, btCmdLen);
            btCmdLen = 0;
        } else if (btCmdLen < sizeof(btCmdBuf) - 1) {
            btCmdBuf[btCmdLen++] = c;
        }
    }
}

void handleBtCommand(char* cmd, uint8_t len)
{
    // 跳过前导空格
    while (len > 0 && (*cmd == ' ' || *cmd == '\t')) { cmd++; len--; }
    if (len == 0) return;
    char key = *cmd; cmd++; len--;
    while (len > 0 && (*cmd == ' ' || *cmd == '\t')) { cmd++; len--; }
    uint8_t vlen = len;
    while (vlen > 0 && (cmd[vlen-1] == ' ' || cmd[vlen-1] == '\r' || cmd[vlen-1] == '\n')) vlen--;
    if (vlen == 0) return;
    char vbuf[32];
    if (vlen >= sizeof(vbuf)) vlen = sizeof(vbuf) - 1;
    for (uint8_t i = 0; i < vlen; i++) vbuf[i] = cmd[i];
    vbuf[vlen] = '\0';

    // 值必须为数字，否则忽略（防 "SAVE"/"DEF" 等非数字串被误当 key+val=0 误设）
    {
        char c0 = vbuf[0];
        if (!((c0>='0'&&c0<='9')||c0=='-'||c0=='.')) return;
    }

    if (key == '?') {
        BTSerial.print("PAR: ");
        BTSerial.print("P="); BTSerial.print(balanceKp, 2);
        BTSerial.print(" D="); BTSerial.print(balanceKd, 2);
        BTSerial.print(" O="); BTSerial.print(balanceAngleOffset, 2);
        BTSerial.print(" Z="); BTSerial.print(BALANCE_ANGLE_DEADBAND_DEG, 2);
        BTSerial.print(" L="); BTSerial.print(BALANCE_OUTPUT_LIMIT_HZ, 1);
        BTSerial.print(" S="); BTSerial.print(BALANCE_OUTPUT_SLEW_HZ, 1);
        BTSerial.print(" F="); BTSerial.print(OUTER_FULL_ANGLE_DEG, 2);
        BTSerial.print(" G="); BTSerial.print(OUTER_DISABLE_ANGLE_DEG, 2);
        BTSerial.print(" T="); BTSerial.print(turnKp, 2);
        BTSerial.print(" U="); BTSerial.print(turnKd, 2);
        BTSerial.print(" K="); BTSerial.print(speedKp, 2);
        BTSerial.print(" I="); BTSerial.print(speedKi, 2);
        BTSerial.print(" N="); BTSerial.print(MAX_SPEED_ANGLE, 2);
        BTSerial.print(" W="); BTSerial.print(SPEED_INTEGRAL_LIMIT, 1);
        BTSerial.println();
        return;
    }

    float val = atof(vbuf);
    switch (key) {
        case 'P': case 'p': balanceKp = val; break;
        case 'D': case 'd': balanceKd = val; break;
        case 'O': case 'o': balanceAngleOffset = val; break;
        case 'Z': case 'z': BALANCE_ANGLE_DEADBAND_DEG = val; break;
        case 'L': case 'l': BALANCE_OUTPUT_LIMIT_HZ = val; break;
        case 'S': case 's': BALANCE_OUTPUT_SLEW_HZ = val; break;
        case 'F': case 'f': OUTER_FULL_ANGLE_DEG = val; break;
        case 'G': case 'g': OUTER_DISABLE_ANGLE_DEG = val; break;
        case 'T': case 't': turnKp = val; break;
        case 'U': case 'u': turnKd = val; break;
        case 'K': case 'k': speedKp = val; break;
        case 'I': case 'i': speedKi = val; break;
        case 'N': case 'n': MAX_SPEED_ANGLE = val; break;
        case 'W': case 'w': SPEED_INTEGRAL_LIMIT = val; break;
        default:
            BTSerial.println("ERR:unknown cmd");
            return;
    }
    BTSerial.print(key);
    BTSerial.print("=");
    BTSerial.print(val, 3);
    BTSerial.println(" OK");
}

void loop()

{

    uint32_t nowUs =

        micros();

  

    // 蓝牙在线调参：每圈都轮询，独立于400Hz时序

    pollBtCommand();

  
  

    // --------------------------------------------------------

    // 没到2.5ms就直接返回

    // --------------------------------------------------------

  

    if (

        (uint32_t)(

            nowUs

            -

            lastControlUs

        )

        <

        CONTROL_PERIOD_US

    )

    {

        return;

    }

  
  

    // --------------------------------------------------------

    // 保持固定时间基准

    // --------------------------------------------------------

  

    lastControlUs +=

        CONTROL_PERIOD_US;

  
  

    // --------------------------------------------------------

    // 读取MPU6050（函数名沿用BMI270）

    // --------------------------------------------------------

  

    if (

        !readBMI270()

    )

    {

        imuFailCount++;

  

        if (

            imuFailCount

            <

            IMU_FAIL_CONFIRM_COUNT

        )

        {

            // 保持上一有效STEP输出

            return;

        }

  

        stopBothMotors();

  

        balanceOutput =

            0.0f;

  

        finalBalanceOutput =

            0.0f;

  

        speedDirectOutput =

            0.0f;

  

        speedDirectAuthority =

            0.0f;

  

        return;

    }

  

    imuFailCount =

        0;

  
  

    // --------------------------------------------------------

    // Kalman姿态

    // --------------------------------------------------------

  

    updateAttitude();

  
  

    // --------------------------------------------------------

    // 多源速度Kalman预测 400Hz

    //

    // IMU负责快速感知“开始加速”的趋势；

    // 编码器会在外环20Hz时进行绝对速度修正。

    // --------------------------------------------------------

  

    predictVehicleSpeedKalman();

  
  

    // --------------------------------------------------------

    // 蓝牙遥测20Hz

    // 放在倒地保护之前，这样即使触发倒地保护，

    // 手机上仍能继续看到IMU/Kalman数据。

    // --------------------------------------------------------

  

    telemetryDivider++;

  

    if (

        telemetryDivider

        >=

        TELEMETRY_DIVIDER

    )

    {

        telemetryDivider =

            0;

  

        sendTelemetry();

    }

  
  

    // --------------------------------------------------------

    // 倒地保护（连续确认，避免单帧尖峰突然停机）

    // --------------------------------------------------------

  

    float fallAngleError =

        fabsf(

            pitch

            -

            balanceAngleOffset

        );

  

    if (

        fallAngleError

        >=

        FALL_ANGLE_DEG

    )

    {

        fallConfirmCount++;

    }

    else

    {

        fallConfirmCount =

            0;

    }

  

    bool confirmedFall =

        fallConfirmCount

        >=

        FALL_CONFIRM_COUNT;

  

    bool emergencyFall =

        fallAngleError

        >=

        EMERGENCY_FALL_ANGLE_DEG;

  

    if (

        confirmedFall

        ||

        emergencyFall

    )

    {

        stopBothMotors();

  

        balanceOutput =

            0.0f;

  

        turnOutput =

            0.0f;

  

        speedError =

            0.0f;

  



        speedDirectOutput =

            0.0f;

  

        speedDirectAuthority =

            0.0f;

  

        finalBalanceOutput =

            0.0f;

  

        outerLoopAuthority =

            0.0f;

  

        motor1ZeroHoldCount =

            0;

  

        motor2ZeroHoldCount =

            0;

  

        targetForwardSpeed =

            0.0f;

  

        noInterrupts();

  

        int32_t resetCount1 =

            encoder1Count;

  

        int32_t resetCount2 =

            encoder2Count;

  

        interrupts();

  

        lastEncoder1Count =

            resetCount1;

  

        lastEncoder2Count =

            resetCount2;

  

        positionZeroCount1 =

            resetCount1;

  

        positionZeroCount2 =

            resetCount2;

  

        vehiclePosition =

            0.0f;

  

        positionDirectOutput =

            0.0f;

  

        vehicleSpeed =

            0.0f;

  

        rawVehicleSpeed =

            0.0f;

  

        imuForwardAccelG =

            0.0f;

  

        imuForwardAccelFilteredG =

            0.0f;

  

        speedKalmanP =

            1.0f;

  

        targetBalanceAngle =

            balanceAngleOffset;

  

        return;

    }

  
  

    // --------------------------------------------------------

    // 速度PI外环20Hz，同时计算速度P直接STEP快速阻尼

    //

    // 400Hz控制循环，每20次执行一次 = 50ms。

    // 50ms编码器窗口用于降低低速时0/1计数跳变。

    // --------------------------------------------------------

  

    outerLoopDivider++;

  
  

    if (

        outerLoopDivider

        >=

        20

    )

    {

        outerLoopDivider =

            0;

  

        updateOuterLoop();

    }

  
  

    // --------------------------------------------------------

    // 外环权限管理400Hz

    // 大角度时速度/位置环平滑退出，直立PD优先

    // --------------------------------------------------------

  

    updateOuterLoopAuthority();

  
  

    // --------------------------------------------------------

    // 位置快速拉回400Hz

    // --------------------------------------------------------

  

    updatePositionHold();

  
  

    // --------------------------------------------------------

    // 直立环400Hz

    // --------------------------------------------------------

  

    updateBalanceLoop();

  
  

    // --------------------------------------------------------

    // 转向环400Hz

    // --------------------------------------------------------

  

    updateTurnLoop();

  
  

    // --------------------------------------------------------

    // 电机合成

    // --------------------------------------------------------

  

    float motor1Command =

        finalBalanceOutput

        +

        turnOutput;

  
  

    float motor2Command =

        finalBalanceOutput

        -

        turnOutput;

  
  

    setMotor1(

        motor1Command

    );

  
  

    setMotor2(

        motor2Command

    );

    // 非阻塞遥测 flush（每控制周期按预算发送缓冲内容）

    flushTelemetryTx();
}
